### Task 4: JSON-LD Product schema on product detail page

**Files:**
- Modify: `frontend/src/app/[locale]/products/[slug]/page.tsx`

**Background:** JSON-LD structured data is injected via a `<script type="application/ld+json">` tag. In Next.js Server Components this is done by rendering a `<script>` element with the serialized JSON — the content comes from server-controlled data (product from DB), never from user HTML. This is the schema.org recommended pattern and is safe.

**Step 1: Read current product detail page**

```bash
cat frontend/src/app/[locale]/products/[slug]/page.tsx
```

**Step 2: Add generateMetadata export**

```tsx
export async function generateMetadata({
  params,
}: {
  params: Promise<{ locale: string; slug: string }>;
}) {
  const { locale, slug } = await params;
  const base = process.env.NEXT_PUBLIC_BASE_URL ?? "https://shopunity.pk";
  let product;
  try {
    product = await productsApi.getBySlug(slug);
  } catch {
    return {};
  }
  return {
    title: `${product.name} — ShopUnity`,
    description: product.description?.slice(0, 160),
    alternates: {
      canonical: `${base}/${locale}/products/${slug}`,
      languages: {
        en: `${base}/en/products/${slug}`,
        ar: `${base}/ar/products/${slug}`,
        ur: `${base}/ur/products/${slug}`,
      },
    },
    openGraph: {
      title: product.name,
      description: product.description?.slice(0, 160),
      images: product.images?.[0] ? [{ url: product.images[0] }] : [],
    },
  };
}
```

**Step 3: Add JSON-LD component inside the page JSX**

At the top of the returned JSX (inside the outer div), add:

```tsx
{/* JSON-LD structured data — server-controlled JSON, safe */}
<script
  type="application/ld+json"
  // NOTE: content is JSON.stringify of server data — not user HTML
  // The exact JSX attribute is: dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }}
  // See Next.js docs: https://nextjs.org/docs/app/building-your-application/optimizing/metadata
/>
```

Full implementation pattern:

```tsx
// Build the jsonLd object from product data
const jsonLd = {
  "@context": "https://schema.org",
  "@type": "Product",
  name: product.name,
  description: product.description ?? "",
  sku: product.sku ?? product.id,
  image: product.images ?? [],
  offers: {
    "@type": "Offer",
    priceCurrency: "PKR",
    price: product.price,
    availability:
      product.stock_qty > 0
        ? "https://schema.org/InStock"
        : "https://schema.org/OutOfStock",
    url: `${base}/${locale}/products/${slug}`,
  },
  ...(product.avg_rating && product.review_count
    ? {
        aggregateRating: {
          "@type": "AggregateRating",
          ratingValue: product.avg_rating,
          reviewCount: product.review_count,
        },
      }
    : {}),
};

// Then render the script tag in JSX with dangerouslySetInnerHTML set to
// { __html: JSON.stringify(jsonLd) }
```

### Task 7: Translate schemas + FastAPI router with Redis cache

**Files:**
- Create: `backend/app/schemas/translate.py`
- Create: `backend/app/routers/translate.py`

**Step 1: Write the schema**

```python
# backend/app/schemas/translate.py
from pydantic import BaseModel, Field


class TranslateRequest(BaseModel):
    text: str = Field(..., max_length=5000)
    target_lang: str = Field(..., pattern=r"^(AR|UR|EN)$")


class TranslateResponse(BaseModel):
    translated_text: str
    cached: bool
```

**Step 2: Write the router**

```python
# backend/app/routers/translate.py
import hashlib
from fastapi import APIRouter, HTTPException
import httpx

from app.core.config import get_settings
from app.core.redis import get_redis
from app.schemas.translate import TranslateRequest, TranslateResponse

router = APIRouter(prefix="/v1/translate", tags=["translate"])

CACHE_TTL = 86400  # 24 hours


@router.post("", response_model=TranslateResponse)
async def translate_text(body: TranslateRequest):
    settings = get_settings()
    if not settings.deepl_api_key:
        raise HTTPException(503, "Translation service not configured")

    content_hash = hashlib.md5(body.text.encode()).hexdigest()
    cache_key = f"translate:{body.target_lang}:{content_hash}"

    redis = await get_redis()
    cached = await redis.get(cache_key)
    if cached:
        return TranslateResponse(translated_text=cached.decode(), cached=True)

    async with httpx.AsyncClient() as client:
        resp = await client.post(
            "https://api-free.deepl.com/v2/translate",
            headers={"Authorization": f"DeepL-Auth-Key {settings.deepl_api_key}"},
            json={"text": [body.text], "target_lang": body.target_lang},
            timeout=10.0,
        )

    if resp.status_code != 200:
        raise HTTPException(502, "Translation service error")

    translated = resp.json()["translations"][0]["text"]
    await redis.setex(cache_key, CACHE_TTL, translated.encode())
    return TranslateResponse(translated_text=translated, cached=False)
```

**Step 1: Write formatPrice utility**

```ts
// frontend/src/lib/format-price.ts

type Locale = "en" | "ar" | "ur";

const LOCALE_CURRENCY: Record<Locale, string> = {
  en: "PKR",
  ar: "PKR",
  ur: "PKR",
};

export function formatPrice(amount: number, locale: Locale = "en"): string {
  const currency = LOCALE_CURRENCY[locale];
  return new Intl.NumberFormat(locale === "en" ? "en-PK" : locale, {
    style: "currency",
    currency,
    maximumFractionDigits: 0,
  }).format(amount);
}
```

**Step 2: Write seed script**

```python
# backend/scripts/seed_currencies.py
"""Seed PKR, USD, AED into the currencies table."""
import asyncio
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
from sqlalchemy.orm import sessionmaker
from sqlalchemy.dialects.postgresql import insert as pg_insert
from app.core.config import get_settings
from app.models.catalog import Currency

CURRENCIES = [
    {"code": "PKR", "name": "Pakistani Rupee", "symbol": "₨", "is_default": True, "exchange_rate_to_pkr": 1.0},
    {"code": "USD", "name": "US Dollar", "symbol": "$", "is_default": False, "exchange_rate_to_pkr": 278.5},
    {"code": "AED", "name": "UAE Dirham", "symbol": "د.إ", "is_default": False, "exchange_rate_to_pkr": 75.8},
]


async def seed():
    settings = get_settings()
    engine = create_async_engine(settings.database_url)
    async_session = sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)
    async with async_session() as session:
        for c in CURRENCIES:
            stmt = pg_insert(Currency).values(**c).on_conflict_do_update(
                index_elements=["code"], set_=c
            )
            await session.execute(stmt)
        await session.commit()
    print("Seeded 3 currencies.")


if __name__ == "__main__":
    asyncio.run(seed())
```
